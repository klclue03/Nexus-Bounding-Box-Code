#include "rclcpp/rclcpp.hpp"
#include "sensor_msgs/msg/point_cloud2.hpp"
#include "visualization_msgs/msg/marker_array.hpp"
#include <algorithm>
#include <cmath>
#include <limits>
#include <pcl_conversions/pcl_conversions.h>
#include "cluster_extraction.hpp"
#include "bounding_box.hpp"

#include "bounding_box.hpp"
#include "../pcl_cluster_extraction/cluster_extraction.hpp"
#include "../pcl_ground_removal/ground_removal.hpp"
#include "../pcl_preprocessor/filter.hpp"

using std::placeholders::_1;

class PointCloudBoundingBox : public rclcpp::Node
{
public:
    PointCloudBoundingBox() : Node("pointcloud_bounding_box")
    {
      // Create Subscriber to "input_pcl" topic
      subscription_ = this->create_subscription<sensor_msgs::msg::PointCloud2>(
        "/unilidar/cloud",
        10,
        std::bind(&PointCloudBoundingBox::pointcloud_callback, this, _1));

      // Create publisher to "output_pcl" topic
      publisher_ = this->create_publisher<sensor_msgs::msg::PointCloud2>(
        "output_pcl", 
        10);

      // Create publisher to "bounding_boxes" topic
      marker_publisher_ = this->create_publisher<visualization_msgs::msg::MarkerArray>(
        "bounding_boxes", 
        10);
      
      // Tuning knobs, exposed as parameters so they can be changed without rebuilding
      double ground_threshold = this->declare_parameter("ground_threshold", 0.05);
      double ground_clearance = this->declare_parameter("ground_clearance", 0.15);
      double cluster_tolerance = this->declare_parameter("cluster_tolerance", 0.075);
      int min_cluster_size = this->declare_parameter("min_cluster_size", 5);
      max_cluster_points_ = this->declare_parameter("max_cluster_size", 2000);

      // Initialize the GroundRemoval class with a threshold for plane segmentation
      ground_removal_ = std::make_shared<GroundRemoval>(ground_threshold, ground_clearance);

      // Initialize the ClusterExtraction class (unbounded here; oversized clusters are reported below)
      cluster_extraction_ = std::make_shared<ClusterExtraction>(
        cluster_tolerance, min_cluster_size, std::numeric_limits<int>::max());

      // Initialize the BoundingBox class
      bounding_box_ = std::make_shared<BoundingBox>();
    }


private:
    void pointcloud_callback(const sensor_msgs::msg::PointCloud2::SharedPtr msg)
    {
        RCLCPP_INFO(this->get_logger(), "Received PointCloud2 with %d data points", msg->width * msg->height);

        try
        {
            // Preprocess and perform ground removal
            sensor_msgs::msg::PointCloud2 preprocessed_msg;
            filter.preprocessPointCloud(msg, preprocessed_msg); // Assuming you have filter defined
            pcl::PointCloud<pcl::PointXYZ>::Ptr cloud(new pcl::PointCloud<pcl::PointXYZ>);
            pcl::fromROSMsg(preprocessed_msg, *cloud);

            // Perform ground removal
            pcl::PointCloud<pcl::PointXYZ>::Ptr cloud_filtered = ground_removal_->removeGround(cloud);
            
            // auto cloud_filtered = cloud;

            // Perform cluster extraction (no upper size limit, so oversized clusters can be reported below)
            auto clusters = cluster_extraction_->extractClusters(cloud_filtered);

            // Create a MarkerArray for bounding boxes, clearing boxes left over from the previous frame
            visualization_msgs::msg::MarkerArray marker_array;
            visualization_msgs::msg::Marker clear_marker;
            clear_marker.header = msg->header;
            clear_marker.action = visualization_msgs::msg::Marker::DELETEALL;
            marker_array.markers.push_back(clear_marker);

            // Debug cloud: accepted clusters get a distinct color each, rejected clusters are red
            pcl::PointCloud<pcl::PointXYZRGB> debug_cloud;
            size_t num_rejected = 0;

            // Process each cluster
            for (size_t i = 0; i < clusters.size(); ++i)
            {
                const auto &cluster = clusters[i];
                Eigen::Vector4f min_pt, max_pt;
                pcl::getMinMax3D(*cluster, min_pt, max_pt);
                Eigen::Vector4f center = (min_pt + max_pt) / 2.0f;
                bool rejected = static_cast<int>(cluster->size()) > max_cluster_points_;

                RCLCPP_INFO(this->get_logger(),
                    "Cluster %zu: %zu points, %.2f m away, size %.2f x %.2f x %.2f m%s",
                    i, cluster->size(), center.head<3>().norm(),
                    max_pt.x() - min_pt.x(), max_pt.y() - min_pt.y(), max_pt.z() - min_pt.z(),
                    rejected ? "  -> REJECTED (too many points)" : "");

                uint8_t r = 255, g = 0, b = 0;
                if (!rejected)
                {
                    // Create a bounding box for each cluster
                    auto marker = bounding_box_->createBoundingBoxMarker(cluster, msg->header, i);
                    marker_array.markers.push_back(marker);

                    // Golden-ratio hue spacing keeps neighbouring cluster colors distinct (and away from pure red)
                    float hue = std::fmod(0.3f + i * 0.618f, 0.75f) + 0.1f;
                    r = static_cast<uint8_t>(255 * std::clamp(std::abs(hue * 6 - 3) - 1, 0.0f, 1.0f));
                    g = static_cast<uint8_t>(255 * std::clamp(2 - std::abs(hue * 6 - 2), 0.0f, 1.0f));
                    b = static_cast<uint8_t>(255 * std::clamp(2 - std::abs(hue * 6 - 4), 0.0f, 1.0f));
                }
                else
                {
                    ++num_rejected;
                }

                for (const auto &p : cluster->points)
                {
                    pcl::PointXYZRGB cp(r, g, b);
                    cp.x = p.x; cp.y = p.y; cp.z = p.z;
                    debug_cloud.push_back(cp);
                }
            }

            RCLCPP_INFO(this->get_logger(), "%zu clusters, %zu rejected", clusters.size(), num_rejected);

            // Publish the colored clusters for inspection in RViz
            sensor_msgs::msg::PointCloud2 debug_msg;
            pcl::toROSMsg(debug_cloud, debug_msg);
            debug_msg.header = msg->header;
            publisher_->publish(debug_msg);

            // Publish the bounding boxes
            marker_publisher_->publish(marker_array);
        }
        catch (const std::runtime_error &e)
        {
            RCLCPP_WARN(this->get_logger(), "Processing failed: %s", e.what());
        }
    }

    rclcpp::Publisher<sensor_msgs::msg::PointCloud2>::SharedPtr publisher_;
    rclcpp::Publisher<visualization_msgs::msg::MarkerArray>::SharedPtr marker_publisher_;
    rclcpp::Subscription<sensor_msgs::msg::PointCloud2>::SharedPtr subscription_;
    PointCloudFilter filter;                                // Filter class object
    std::shared_ptr<GroundRemoval> ground_removal_;         // Assuming GroundRemoval is defined elsewhere
    std::shared_ptr<ClusterExtraction> cluster_extraction_; // ClusterExtraction instance
    std::shared_ptr<BoundingBox> bounding_box_;             // BoundingBox instance
    int max_cluster_points_ = 2500;                         // Clusters larger than this get no bounding box
};

int main(int argc, char **argv)
{
    rclcpp::init(argc, argv);
    rclcpp::spin(std::make_shared<PointCloudBoundingBox>());
    rclcpp::shutdown();
    return 0;
}
